Use it

<div id="dusty" style="width:160px"></div>
<script type="module">
import { mountDusty } from './dusty-avatar.js'
const def = await fetch('./dusty.avatar.json')
  .then(r => r.json())
const dusty = mountDusty(
  document.getElementById('dusty'), def)
dusty.play('proud')      // or 'idle', 'cleanup'…
dusty.pose('sleepy', 400) // one pose, 400 ms fade
dusty.set({ gaze: [2, 0], color: '#9FD8C9' })
</script>
